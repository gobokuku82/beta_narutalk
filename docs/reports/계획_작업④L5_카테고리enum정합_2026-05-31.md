# 1. 변환 스크립트 작성 (repo 에 commit X, 한 회용)
cat > /tmp/convert_yaml_category.py <<'EOF'
import re
from pathlib import Path
CAT_DIR = Path("backend/app/dream_agent/tools/catalog")
changed = []
for yml in CAT_DIR.rglob("*.yaml"):
    if yml.name.startswith("_"):
        continue
    parts = yml.relative_to(CAT_DIR).parts
    category = parts[0]  # collection / normalization / ...
    text = yml.read_text(encoding="utf-8")
    new_text = re.sub(r"^category:\s*\S+", f"category: {category}",
                      text, count=1, flags=re.MULTILINE)
    if new_text != text:
        yml.write_text(new_text, encoding="utf-8")
        changed.append((str(yml.relative_to(CAT_DIR)), category))
print(f"Changed {len(changed)} files:")
for path, cat in sorted(changed):
    print(f"  {path} → {cat}")
EOF

# 2. 실행
